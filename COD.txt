#include "main.h"
#include <stdio.h>
#include <string.h>
#include <stdbool.h>

/* ===================== DEFINES ===================== */

#define MAX3010X_ADDR   (0x57 << 1)
#define LCD_ADDR        (0x27 << 1)

/* MAX3010x registers */
#define REG_INTR_ENABLE_1   0x02
#define REG_INTR_ENABLE_2   0x03
#define REG_FIFO_WR_PTR     0x04
#define REG_OVF_COUNTER     0x05
#define REG_FIFO_RD_PTR     0x06
#define REG_FIFO_DATA       0x07
#define REG_FIFO_CONFIG     0x08
#define REG_MODE_CONFIG     0x09
#define REG_SPO2_CONFIG     0x0A
#define REG_LED1_PA         0x0C
#define REG_LED2_PA         0x0D

/* thresholds */
#define FINGER_THRESHOLD        8000
#define BPM_LOW_ALARM          50
#define BPM_HIGH_ALARM         120
#define SPO2_LOW_ALARM         90

/* peak detector tuning */
#define PEAK_THRESHOLD_OFFSET   300

/* separate windows */
#define BPM_WINDOW_SIZE         50
#define SPO2_WINDOW_SIZE        20

/* intervals */
#define PEAK_INTERVAL           20      /* ms: fast pulse reading */
#define SPO2_INTERVAL           50     /* ms: SpO2 sample cooldown */
#define LCD_REFRESH_INTERVAL    1000     /* ms: LCD refresh rate */

/* ===================== HANDLES ===================== */

I2C_HandleTypeDef hi2c1;
UART_HandleTypeDef huart1;

/* ===================== PROTOTYPES ===================== */

void SystemClock_Config(void);
static void MX_GPIO_Init(void);
static void MX_I2C1_Init(void);
static void MX_USART1_UART_Init(void);
void Error_Handler(void);

/* LCD */
void lcd_send_cmd(uint8_t cmd);
void lcd_send_data(uint8_t data);
void lcd_init(void);
void lcd_set_cursor(uint8_t row, uint8_t col);
void lcd_print_line(uint8_t row, const char *text);

/* Bluetooth / UART */
void bt_send(const char *msg);

/* MAX3010x */
void max_write(uint8_t reg, uint8_t val);
void max_init(void);
void max_read(uint32_t *red, uint32_t *ir);

/* helpers */
int clamp_int(int v, int vmin, int vmax);

/* ===================== LCD ===================== */

void lcd_send_cmd(uint8_t cmd)
{
    uint8_t data_t[4];

    data_t[0] = (cmd & 0xF0) | 0x0C;
    data_t[1] = (cmd & 0xF0) | 0x08;
    data_t[2] = ((cmd << 4) & 0xF0) | 0x0C;
    data_t[3] = ((cmd << 4) & 0xF0) | 0x08;

    HAL_I2C_Master_Transmit(&hi2c1, LCD_ADDR, data_t, 4, HAL_MAX_DELAY);
}

void lcd_send_data(uint8_t data)
{
    uint8_t data_t[4];

    data_t[0] = (data & 0xF0) | 0x0D;
    data_t[1] = (data & 0xF0) | 0x09;
    data_t[2] = ((data << 4) & 0xF0) | 0x0D;
    data_t[3] = ((data << 4) & 0xF0) | 0x09;

    HAL_I2C_Master_Transmit(&hi2c1, LCD_ADDR, data_t, 4, HAL_MAX_DELAY);
}

void lcd_init(void)
{
    HAL_Delay(50);

    lcd_send_cmd(0x30);
    HAL_Delay(5);
    lcd_send_cmd(0x30);
    HAL_Delay(1);
    lcd_send_cmd(0x30);
    HAL_Delay(10);

    lcd_send_cmd(0x20);
    HAL_Delay(10);

    lcd_send_cmd(0x28);
    lcd_send_cmd(0x08);
    lcd_send_cmd(0x01);
    HAL_Delay(3);
    lcd_send_cmd(0x06);
    lcd_send_cmd(0x0C);
}

void lcd_set_cursor(uint8_t row, uint8_t col)
{
    uint8_t pos = (row == 0) ? (0x80 + col) : (0xC0 + col);
    lcd_send_cmd(pos);
}

void lcd_print_line(uint8_t row, const char *text)
{
    char buf[17];
    int i;

    for (i = 0; i < 16; i++)
        buf[i] = ' ';

    buf[16] = '\0';

    for (i = 0; i < 16 && text[i] != '\0'; i++)
        buf[i] = text[i];

    lcd_set_cursor(row, 0);

    for (i = 0; i < 16; i++)
        lcd_send_data((uint8_t)buf[i]);
}

/* ===================== BLUETOOTH / UART ===================== */

void bt_send(const char *msg)
{
    HAL_UART_Transmit(&huart1, (uint8_t*)msg, strlen(msg), HAL_MAX_DELAY);
}

/* ===================== MAX3010x ===================== */

void max_write(uint8_t reg, uint8_t val)
{
    uint8_t data[2];

    data[0] = reg;
    data[1] = val;

    HAL_I2C_Master_Transmit(&hi2c1, MAX3010X_ADDR, data, 2, HAL_MAX_DELAY);
}

void max_init(void)
{
    max_write(REG_MODE_CONFIG, 0x40);
    HAL_Delay(100);

    max_write(REG_INTR_ENABLE_1, 0x00);
    max_write(REG_INTR_ENABLE_2, 0x00);

    max_write(REG_FIFO_WR_PTR, 0x00);
    max_write(REG_OVF_COUNTER, 0x00);
    max_write(REG_FIFO_RD_PTR, 0x00);

    max_write(REG_FIFO_CONFIG, 0x0F);
    max_write(REG_MODE_CONFIG, 0x03);
    max_write(REG_SPO2_CONFIG, 0x27);
    max_write(REG_LED1_PA, 0x24);
    max_write(REG_LED2_PA, 0x24);

    HAL_Delay(100);
}

void max_read(uint32_t *red, uint32_t *ir)
{
    uint8_t reg = REG_FIFO_DATA;
    uint8_t data[6];

    HAL_I2C_Master_Transmit(&hi2c1, MAX3010X_ADDR, &reg, 1, HAL_MAX_DELAY);
    HAL_I2C_Master_Receive(&hi2c1, MAX3010X_ADDR, data, 6, HAL_MAX_DELAY);

    *red = (((uint32_t)data[0] << 16) |
            ((uint32_t)data[1] << 8)  |
            data[2]) & 0x3FFFF;

    *ir  = (((uint32_t)data[3] << 16) |
            ((uint32_t)data[4] << 8)  |
            data[5]) & 0x3FFFF;
}

/* ===================== HELPERS ===================== */

int clamp_int(int v, int vmin, int vmax)
{
    if (v < vmin) return vmin;
    if (v > vmax) return vmax;
    return v;
}

/* ===================== MAIN ===================== */

int main(void)
{
    uint32_t red = 0;
    uint32_t ir = 0;

    /* SpO2 raw ring buffer */
    uint32_t red_buf[SPO2_WINDOW_SIZE] = {0};
    uint32_t ir_buf[SPO2_WINDOW_SIZE]  = {0};
    uint8_t spo2_head = 0;
    uint8_t spo2_count = 0;

    /* BPM ring buffer */
    int bpm_buf[BPM_WINDOW_SIZE] = {0};
    uint8_t bpm_head = 0;
    uint8_t bpm_count = 0;

    /* calculated display values */
    int bpm_disp = 0;
    int spo2_disp = 0;

    /* peak detection state */
    uint32_t prev_ir1 = 0;
    uint32_t prev_ir2 = 0;
    uint32_t ir_avg = 0;
    uint32_t last_beat_tick = 0;

    /* timers */
    uint32_t last_read_tick = 0;
    uint32_t last_spo2_tick = 0;
    uint32_t last_lcd_tick = 0;
    uint32_t last_bt_tick = 0;

    bool finger_present = false;

    char line1[17] = {0};
    char line2[17] = {0};
    char last_line1[17] = {0};
    char last_line2[17] = {0};
    char bt_msg[64];

    HAL_Init();
    SystemClock_Config();

    MX_GPIO_Init();
    MX_I2C1_Init();
    MX_USART1_UART_Init();

    HAL_Delay(500);

    lcd_init();
    max_init();

    lcd_print_line(0, "Sistem pornit");
    lcd_print_line(1, "Asteapta...");
    HAL_Delay(1000);

    while (1)
    {
        uint32_t now = HAL_GetTick();

        bool do_read = (now - last_read_tick >= PEAK_INTERVAL);
        bool do_spo2 = (now - last_spo2_tick >= SPO2_INTERVAL);
        bool do_lcd  = (now - last_lcd_tick  >= LCD_REFRESH_INTERVAL);

        if (!do_read && !do_spo2 && !do_lcd)
            continue;

        /* ============================================================
           FAST SENSOR READ + BPM PEAK DETECTION
           Runs every PEAK_INTERVAL ms
           ============================================================ */
        if (do_read)
        {
            last_read_tick = now;

            max_read(&red, &ir);

            if (ir < FINGER_THRESHOLD)
            {
                finger_present = false;

                HAL_GPIO_WritePin(GPIOB, GPIO_PIN_5, GPIO_PIN_RESET);

                spo2_head = 0;
                spo2_count = 0;

                bpm_head = 0;
                bpm_count = 0;

                bpm_disp = 0;
                spo2_disp = 0;

                prev_ir1 = 0;
                prev_ir2 = 0;
                ir_avg = 0;
                last_beat_tick = 0;
            }
            else
            {
                finger_present = true;

                if (ir_avg == 0)
                    ir_avg = ir;
                else
                    ir_avg = (ir_avg * 7 + ir) / 8;

                if (prev_ir1 > prev_ir2 &&
                    prev_ir1 > ir &&
                    prev_ir1 > (ir_avg + PEAK_THRESHOLD_OFFSET))
                {
                    if (last_beat_tick != 0)
                    {
                        uint32_t dt = now - last_beat_tick;

                        if (dt > 300 && dt < 1500)
                        {
                            int bpm = (int)(60000u / dt);

                            if (bpm >= 40 && bpm <= 180)
                            {
                                int sum = 0;

                                bpm_buf[bpm_head] = bpm;
                                bpm_head = (bpm_head + 1) % BPM_WINDOW_SIZE;

                                if (bpm_count < BPM_WINDOW_SIZE)
                                    bpm_count++;

                                for (uint8_t k = 0; k < bpm_count; k++)
                                    sum += bpm_buf[k];

                                bpm_disp = sum / bpm_count;
                            }
                        }
                    }

                    last_beat_tick = now;
                }

                prev_ir2 = prev_ir1;
                prev_ir1 = ir;
            }
        }

        /* ============================================================
           SpO2 SAMPLING WINDOW
           Runs every SPO2_INTERVAL ms
           ============================================================ */
        if (do_spo2)
        {
            last_spo2_tick = now;

            if (finger_present)
            {
                red_buf[spo2_head] = red;
                ir_buf[spo2_head] = ir;

                spo2_head = (spo2_head + 1) % SPO2_WINDOW_SIZE;

                if (spo2_count < SPO2_WINDOW_SIZE)
                    spo2_count++;

                if (spo2_count == SPO2_WINDOW_SIZE)
                {
                    uint32_t r_min = 0xFFFFFFFF;
                    uint32_t r_max = 0;
                    uint32_t i_min = 0xFFFFFFFF;
                    uint32_t i_max = 0;

                    uint64_t r_sum = 0;
                    uint64_t i_sum = 0;

                    for (uint8_t k = 0; k < SPO2_WINDOW_SIZE; k++)
                    {
                        if (red_buf[k] < r_min) r_min = red_buf[k];
                        if (red_buf[k] > r_max) r_max = red_buf[k];

                        if (ir_buf[k] < i_min) i_min = ir_buf[k];
                        if (ir_buf[k] > i_max) i_max = ir_buf[k];

                        r_sum += red_buf[k];
                        i_sum += ir_buf[k];
                    }

                    float r_dc = (float)r_sum / SPO2_WINDOW_SIZE;
                    float i_dc = (float)i_sum / SPO2_WINDOW_SIZE;
                    float r_ac = (float)(r_max - r_min);
                    float i_ac = (float)(i_max - i_min);

                    if (r_dc > 1.0f &&
                        i_dc > 1.0f &&
                        r_ac > 1.0f &&
                        i_ac > 1.0f)
                    {
                        float R = (r_ac / r_dc) / (i_ac / i_dc);
                        int spo2 = (int)(110.0f - 25.0f * R);

                        spo2_disp = clamp_int(spo2, 80, 100);
                    }
                }
            }
        }

        /* ============================================================
           LCD REFRESH + ALARM
           Runs every LCD_REFRESH_INTERVAL ms
           ============================================================ */
        if (do_lcd)
        {
            bool alarm = false;

            last_lcd_tick = now;

            if (!finger_present)
            {
                snprintf(line1, sizeof(line1), "Pune degetul");
                snprintf(line2, sizeof(line2), "pe senzor");

                HAL_GPIO_WritePin(GPIOB, GPIO_PIN_5, GPIO_PIN_RESET);
            }
            else
            {
                if (bpm_count == 0)
                    snprintf(line1, sizeof(line1), "Puls:--- BPM");
                else
                    snprintf(line1, sizeof(line1), "Puls:%3d BPM", bpm_disp);

                if (spo2_count < SPO2_WINDOW_SIZE || spo2_disp == 0)
                    snprintf(line2, sizeof(line2), "SpO2:--- %%");
                else
                    snprintf(line2, sizeof(line2), "SpO2:%3d %%", spo2_disp);

                if (spo2_disp > 0 && spo2_disp < SPO2_LOW_ALARM)
                    alarm = true;

                if (bpm_disp > 0 &&
                    (bpm_disp < BPM_LOW_ALARM || bpm_disp > BPM_HIGH_ALARM))
                    alarm = true;

                if (alarm)
                {
                    HAL_GPIO_WritePin(GPIOB, GPIO_PIN_5, GPIO_PIN_SET);

                    if (now - last_bt_tick > 3000)
                    {
                        snprintf(bt_msg, sizeof(bt_msg),
                                 "ALERTA! BPM=%d SpO2=%d\r\n",
                                 bpm_disp,
                                 spo2_disp);

                        bt_send(bt_msg);
                        last_bt_tick = now;
                    }
                }
                else
                {
                    HAL_GPIO_WritePin(GPIOB, GPIO_PIN_5, GPIO_PIN_RESET);
                }
            }

            if (strcmp(line1, last_line1) != 0)
            {
                lcd_print_line(0, line1);
                memcpy(last_line1, line1, sizeof(line1));
            }

            if (strcmp(line2, last_line2) != 0)
            {
                lcd_print_line(1, line2);
                memcpy(last_line2, line2, sizeof(line2));
            }
        }
    }
}

/* ===================== INIT FUNCTIONS ===================== */

static void MX_I2C1_Init(void)
{
    hi2c1.Instance = I2C1;
    hi2c1.Init.Timing = 0x00100D14;
    hi2c1.Init.OwnAddress1 = 0;
    hi2c1.Init.AddressingMode = I2C_ADDRESSINGMODE_7BIT;
    hi2c1.Init.DualAddressMode = I2C_DUALADDRESS_DISABLE;
    hi2c1.Init.OwnAddress2 = 0;
    hi2c1.Init.OwnAddress2Masks = I2C_OA2_NOMASK;
    hi2c1.Init.GeneralCallMode = I2C_GENERALCALL_DISABLE;
    hi2c1.Init.NoStretchMode = I2C_NOSTRETCH_DISABLE;

    if (HAL_I2C_Init(&hi2c1) != HAL_OK)
        Error_Handler();

    if (HAL_I2CEx_ConfigAnalogFilter(&hi2c1, I2C_ANALOGFILTER_ENABLE) != HAL_OK)
        Error_Handler();

    if (HAL_I2CEx_ConfigDigitalFilter(&hi2c1, 0) != HAL_OK)
        Error_Handler();
}

static void MX_USART1_UART_Init(void)
{
    huart1.Instance = USART1;
    huart1.Init.BaudRate = 9600;
    huart1.Init.WordLength = UART_WORDLENGTH_8B;
    huart1.Init.StopBits = UART_STOPBITS_1;
    huart1.Init.Parity = UART_PARITY_NONE;
    huart1.Init.Mode = UART_MODE_TX_RX;
    huart1.Init.HwFlowCtl = UART_HWCONTROL_NONE;
    huart1.Init.OverSampling = UART_OVERSAMPLING_16;
    huart1.Init.OneBitSampling = UART_ONE_BIT_SAMPLE_DISABLE;
    huart1.AdvancedInit.AdvFeatureInit = UART_ADVFEATURE_NO_INIT;

    if (HAL_UART_Init(&huart1) != HAL_OK)
        Error_Handler();
}

static void MX_GPIO_Init(void)
{
    GPIO_InitTypeDef GPIO_InitStruct = {0};

    __HAL_RCC_GPIOA_CLK_ENABLE();
    __HAL_RCC_GPIOB_CLK_ENABLE();

    HAL_GPIO_WritePin(GPIOB, GPIO_PIN_5, GPIO_PIN_RESET);

    GPIO_InitStruct.Pin = GPIO_PIN_5;
    GPIO_InitStruct.Mode = GPIO_MODE_OUTPUT_PP;
    GPIO_InitStruct.Pull = GPIO_NOPULL;
    GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_LOW;

    HAL_GPIO_Init(GPIOB, &GPIO_InitStruct);
}

void SystemClock_Config(void)
{
    RCC_OscInitTypeDef RCC_OscInitStruct = {0};
    RCC_ClkInitTypeDef RCC_ClkInitStruct = {0};

    if (HAL_PWREx_ControlVoltageScaling(PWR_REGULATOR_VOLTAGE_SCALE1) != HAL_OK)
        Error_Handler();

    RCC_OscInitStruct.OscillatorType = RCC_OSCILLATORTYPE_MSI;
    RCC_OscInitStruct.MSIState = RCC_MSI_ON;
    RCC_OscInitStruct.MSICalibrationValue = 0;
    RCC_OscInitStruct.MSIClockRange = RCC_MSIRANGE_6;
    RCC_OscInitStruct.PLL.PLLState = RCC_PLL_NONE;

    if (HAL_RCC_OscConfig(&RCC_OscInitStruct) != HAL_OK)
        Error_Handler();

    RCC_ClkInitStruct.ClockType = RCC_CLOCKTYPE_HCLK | RCC_CLOCKTYPE_SYSCLK
                                | RCC_CLOCKTYPE_PCLK1 | RCC_CLOCKTYPE_PCLK2;
    RCC_ClkInitStruct.SYSCLKSource = RCC_SYSCLKSOURCE_MSI;
    RCC_ClkInitStruct.AHBCLKDivider = RCC_SYSCLK_DIV1;
    RCC_ClkInitStruct.APB1CLKDivider = RCC_HCLK_DIV1;
    RCC_ClkInitStruct.APB2CLKDivider = RCC_HCLK_DIV1;

    if (HAL_RCC_ClockConfig(&RCC_ClkInitStruct, FLASH_LATENCY_0) != HAL_OK)
        Error_Handler();
}

void Error_Handler(void)
{
    __disable_irq();

    while (1)
    {
    }
}
